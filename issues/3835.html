Reproduction test

import { createFakePluginHost } from "@get-bb/plugin-sdk/testing";
import { expect, it } from "vitest";
import plugin from "./server.js";

it("exposes JSON value type constraints to provider schema consumers", async () => {
  const { bb, harness } = createFakePluginHost({
    pluginId: "workflows",
    agentSkillIds: ["workflows"],
  });
  try {
    await plugin(bb);
    const failures: string[] = [];
    for (const [name, property] of [
      ["bb_workflow_run", "args"],
      ["bb_workflow_result", "value"],
    ]) {
      const tool = harness.registrations.agentTools.find((entry) => entry.name === name);
      expect(tool).toBeDefined();
      const schema = tool!.inputSchema;
      if (typeof schema !== "object" || schema === null || !("properties" in schema)) {
        throw new Error("Missing object properties");
      }
      const properties = schema.properties;
      if (typeof properties !== "object" || properties === null || Array.isArray(properties)) {
        throw new Error("Invalid properties");
      }
      const node = Reflect.get(properties, property);
      console.log(JSON.stringify({ tool: name, property, node }));
      if (!node || !("type" in node || "anyOf" in node || "oneOf" in node || "allOf" in node || "$ref" in node)) {
        failures.push(`${name}.${property}`);
      }
    }
    expect(failures, "Provider-facing parameters missing JSON type constraints").toEqual([]);
  } finally {
    await harness.dispose();
  }
});

5. Root cause

The workflow plugin uses one unknown-valued schema for args and value. Tool registration converts the supplied Zod schema using input-mode JSON Schema conversion. The emitted args node has description and default; value has description only. Neither has a type constraint. Pi maps an untyped leaf to Type.Unknown. A provider requiring an explicit type at this position can reject the request before execution. This final remote rejection remains a hypothesis supported by the report, not a live result from this investigation.

6. Proposed fix and automation decision

Define a provider-compatible representation of arbitrary JSON and verify both nested containers and scalar values against the actual endpoint. Keep accepted JSON behavior and recursion compatibility covered. No production fix or PR was created: changing the exposed tool parameter schema violates this automation rule’s simple-fix criteria. A local top-level type assertion alone would not prove full strict-provider compatibility.
