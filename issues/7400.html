1. TL;DR

The report concerns native Node ESM evaluation of an older published SDK, not a running BB host artifact. Current trusted main builds SDK 0.6.138, whose packed root, host, host-testing, and provider-bridge entries all import successfully. The historical defect was bundling the CommonJS dependency cross-spawn into ESM without a usable require function. Main already leaves that declared dependency external, allowing Node to load its CommonJS implementation normally. This repair and runtime-import coverage landed in 9158444cceee through PR #4178 on September 23, 2026, with SDK source version 0.5.21; this investigation does not certify any registry tarball.

import assert from "node:assert/strict";
import { execFileSync, spawnSync } from "node:child_process";
import { mkdtemp, mkdir, readFile, readdir, rm, symlink, writeFile } from "node:fs/promises";
import { createRequire } from "node:module";
import os from "node:os";
import path from "node:path";
import { pathToFileURL } from "node:url";

const checkout = path.resolve(process.argv[2]);
const packageRoot = path.join(checkout, "packages/plugin-sdk");
const manifest = JSON.parse(await readFile(path.join(packageRoot, "package.json"), "utf8"));
const temporary = await mkdtemp(path.join(os.tmpdir(), "sdk-runtime-check-"));
const requireFromSdk = createRequire(path.join(packageRoot, "package.json"));
const { build } = requireFromSdk("esbuild");
const { runtimeBuildOptions, runtimeEntries } = await import(
  pathToFileURL(path.join(packageRoot, "scripts/runtime-entries.mjs")).href
);

try {
  execFileSync("pnpm", ["pack", "--pack-destination", temporary], {
    cwd: packageRoot,
    encoding: "utf8",
    stdio: "pipe",
    maxBuffer: 32 * 1024 * 1024,
  });
  const archives = (await readdir(temporary)).filter((name) => name.endsWith(".tgz"));
  assert.equal(archives.length, 1);
  const consumer = path.join(temporary, "consumer");
  const scope = path.join(consumer, "node_modules/@get-bb");
  await mkdir(scope, { recursive: true });
  execFileSync("tar", ["-xzf", path.join(temporary, archives[0]), "-C", scope]);
  const extracted = path.join(scope, "package");
  const installed = path.join(scope, "plugin-sdk");
  await symlink(extracted, installed, "dir");
  await symlink(path.join(packageRoot, "node_modules"), path.join(extracted, "node_modules"), "dir");
  const packed = JSON.parse(await readFile(path.join(extracted, "package.json"), "utf8"));
  assert.equal(packed.version, manifest.version);
  assert.ok(packed.dependencies["cross-spawn"]);
  console.log(`runtime=${process.version} sdk=${packed.version}`);
  console.log("pack: PASS (normal prepack; frozen-installed dependencies reused)");
  const probes = [
    ["", "experimental_defineHostEntry"],
    ["/host", "experimental_spawnPortableOutputProcess"],
    ["/testing/host", "experimental_createHostEntryHarness"],
    ["/provider-bridge", null],
  ];
  for (const [suffix, symbol] of probes) {
    const program = [
      'import assert from "node:assert/strict";',
      `const loaded = await import(${JSON.stringify(manifest.name + suffix)});`,
      symbol
        ? `assert.equal(typeof loaded[${JSON.stringify(symbol)}], "function");`
        : "assert.ok(Object.keys(loaded).length > 0);",
      `console.log(${JSON.stringify(manifest.name + suffix + ": PASS")});`,
    ].join("\n");
    const probe = path.join(consumer, "probe.mjs");
    await writeFile(probe, program);
    const result = spawnSync(process.execPath, [probe], { cwd: consumer, encoding: "utf8" });
    assert.equal(result.status, 0, result.stderr);
    process.stdout.write(result.stdout);
  }
  const entry = runtimeEntries(manifest.exports).find((candidate) => candidate.subpath === "./host");
  assert.ok(entry);
  const diagnostic = path.join(packageRoot, `.diagnostic-${path.basename(temporary)}.mjs`);
  try {
    await build({
      ...runtimeBuildOptions(packageRoot, entry, diagnostic),
      external: entry.external.filter((name) => name !== "cross-spawn"),
      logLevel: "silent",
    });
    const result = spawnSync(process.execPath, [diagnostic], { encoding: "utf8" });
    assert.notEqual(result.status, 0);
    const failure = result.stderr.match(/^Error: Dynamic require of "(?:path|child_process)" is not supported$/m);
    assert.ok(failure, result.stderr);
    assert.match(result.stderr, /cross-spawn/);
    console.log(`diagnostic (cross-spawn externalization removed): ${failure[0]}`);
    console.log("diagnostic: EXPECTED FAILURE confirms the bundling mechanism");
  } finally {
    await rm(diagnostic, { force: true });
  }
} finally {
  await rm(temporary, { recursive: true, force: true });
}

5. Root cause

The host export map routes native ESM consumers to dist/host.js. The host entry re-exports process utilities; their host-process implementation imports cross-spawn and its parser. These are CommonJS modules, and the frozen-installed cross-spawn entry uses require('child_process'); its parser also requires Node builtins. An ESM bundle that inlines this graph receives esbuild's require helper, which throws when no native require exists. This happens during import, before a plugin handler needs to run.

The existing runtime entry configuration includes:

const PACKAGE_EXTERNALS = ["zod", "zod/*", "cross-spawn"];

The shared build options apply that external list to an ESM Node bundle. The runtime dependencies declare cross-spawn, so Node resolves it through its supported CommonJS interop rather than evaluating inlined require calls. The require banner elsewhere in that configuration belongs to a separate worker artifact, not this repair.

The root index entry already re-exports host-contract. Switching a narrow consumer to that entry avoids the dependency graph, but is not the upstream repair for consumers that need the host-only functions.

The existing runtime-entry regression suite builds published entries, checks dependency declarations, and imports host and provider-bridge in fresh Node processes. All four tests passed here. It does not itself unpack a tarball; the additional inline check above covers that distinction for this investigation.
