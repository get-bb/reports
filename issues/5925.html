// @vitest-environment jsdom

import { afterEach, expect, it, vi } from "vitest";
import { carryUtmToDownloadLinks } from "./download-utm";

afterEach(() => {
  vi.unstubAllGlobals();
  sessionStorage.clear();
  history.replaceState(null, "", "/");
  document.body.replaceChildren();
});

it.each([true, false])("client attribution with GPC=%s", (enabled) => {
  vi.stubGlobal("navigator", { globalPrivacyControl: enabled });
  history.replaceState(
    null,
    "",
    "/?utm_source=regression&utm_campaign=consent&gclid=alpha&gbraid=beta&wbraid=gamma",
  );
  const link = document.createElement("a");
  link.href = "/download/linux";
  document.body.appendChild(link);
  const stop = carryUtmToDownloadLinks();
  try {
    link.dispatchEvent(new MouseEvent("pointerdown", { bubbles: true }));
    const surfaces = {
      landing: new URL(location.href).searchParams,
      storage: new URLSearchParams(
        sessionStorage.getItem("bb:landing-utm") ?? "",
      ),
      download: new URL(link.href).searchParams,
    };
    console.log(
      JSON.stringify({
        enabled,
        surfaces: Object.fromEntries(
          Object.entries(surfaces).map(([name, params]) => [
            name,
            params.toString(),
          ]),
        ),
      }),
    );
    for (const [surface, params] of Object.entries(surfaces)) {
      expect.soft(params.get("utm_source"), surface).toBe("regression");
      expect.soft(params.get("utm_campaign"), surface).toBe("consent");
      for (const [name, value] of [
        ["gclid", "alpha"],
        ["gbraid", "beta"],
        ["wbraid", "gamma"],
      ]) {
        expect
          .soft(params.get(name), `${surface}.${name}`)
          .toBe(enabled ? null : value);
      }
    }
  } finally {
    stop();
  }
});

import { afterEach, expect, it, vi } from "vitest";
import { handleDownload } from "./endpoints";

afterEach(() => vi.unstubAllGlobals());

it.each([true, false])("server attribution with GPC=%s", async (enabled) => {
  const capture = vi.fn(
    async (_input: RequestInfo | URL, _init?: RequestInit) =>
      new Response("{}"),
  );
  vi.stubGlobal("fetch", capture);
  const pending: Promise<void>[] = [];
  const query =
    "utm_source=regression&utm_campaign=consent&gclid=alpha&gbraid=beta&wbraid=gamma";
  const request = new Request(
    `https://web.test/download/linux?instructions=1&${query}`,
    {
      headers: {
        "Sec-GPC": enabled ? "1" : "0",
        referer: `https://web.test/?${query}`,
      },
    },
  );
  const response = await handleDownload(
    "linux",
    request,
    { LANDING_POSTHOG_KEY: "synthetic-test-key" },
    (promise) => pending.push(promise),
  );
  await Promise.all(pending);
  expect(response.status).toBe(302);
  expect(capture).toHaveBeenCalledTimes(1);
  const body = capture.mock.calls[0]?.[1]?.body;
  if (typeof body !== "string") throw new Error("Missing capture body");
  const payload: unknown = JSON.parse(body);
  const redirect = new URL(response.headers.get("location") ?? "", request.url)
    .searchParams;
  console.log(
    JSON.stringify({ enabled, redirect: redirect.toString(), payload }),
  );
  expect.soft(redirect.get("utm_source")).toBe("regression");
  expect.soft(payload).toHaveProperty("properties.utm_campaign", "consent");
  for (const [name, value] of [
    ["gclid", "alpha"],
    ["gbraid", "beta"],
    ["wbraid", "gamma"],
  ]) {
    expect
      .soft(redirect.get(name), `redirect.${name}`)
      .toBe(enabled ? null : value);
    if (enabled) {
      expect.soft(payload).not.toHaveProperty(`properties.${name}`);
    } else {
      expect.soft(payload).toHaveProperty(`properties.${name}`, value);
    }
  }
  const expectedQuery = enabled
    ? "utm_source=regression&utm_campaign=consent"
    : query;
  expect
    .soft(payload)
    .toHaveProperty(
      "properties.$current_url",
      `https://web.test/download/linux?instructions=1&${expectedQuery}`,
    );
  expect
    .soft(payload)
    .toHaveProperty(
      "properties.$referrer",
      `https://web.test/?${expectedQuery}`,
    );
});

5. Root cause

The campaign allowlist includes all three identifiers, but every privacy-filtering path recognizes only one. The client therefore removes gclid yet keeps the other two in the address bar, persisted attribution and rewritten download link. The server repeats the same incomplete policy for install redirects and event URL/referrer sanitization. Because the remaining names are still valid campaign parameters, they also become explicit event properties.

  • Campaign allowlist includes gclid, gbraid and wbraid.
  • Client landing URL and storage remove only gclid; download-link sanitization repeats the same single-name deletion.
  • Install redirect skips only the gclid campaign parameter under Sec-GPC: 1.
  • Event current URL and referrer remove only gclid, then campaign properties are assembled.
  • Existing endpoint tests explicitly expect both braid parameters to remain even under GPC. The three existing test files pass all 40 cases on the original base, explaining why those checks did not detect the gap.
Original client policy:
saved.delete("gclid");
params.delete("gclid");

Original install redirect policy:
if (name === "gclid" && request.headers.get("Sec-GPC") === "1") {
  continue;
}

Original event policy:
requestUrl.searchParams.delete("gclid");
referrerUrl.searchParams.delete("gclid");

6. Proposed fix and disposition

Use one explicit click-identifier list for the existing client and server GPC filters, and remove each listed name only when GPC is enabled. Leave the campaign allowlist and GPC-off behavior intact. Extend the existing behavioral tests across all identifiers and URL/persistence/event surfaces. This is now implemented by the merged #5926; this investigation makes no production change.
