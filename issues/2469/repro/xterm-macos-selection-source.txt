export function rightClickHandler(..., shouldSelectWord: boolean): void {
  moveTextAreaUnderMouseCursor(ev, textarea, screenElement);

  if (shouldSelectWord) {
    selectionService.rightClickSelect(ev);
  }

  textarea.value = selectionService.selectionText;
  textarea.select();
}
