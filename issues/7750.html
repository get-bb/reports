1. TL;DR

The report describes expensive warm thread navigation dominated by style recalculation. A real-browser reproduction was attempted twice, but this host has no Chrome/Chromium, so no browser trace or millisecond measurement was obtained. A smaller diagnostic using the actual timeline and scroll components ran in two clean trusted checkouts and confirmed that remounting repeats geometry and table-related style reads even with the same row objects. It also found that 150- and 300-row histories both mount only 17 top-level timeline rows in this fixture. These findings identify reachable profiling targets, not proof of the reported latency, a whole-timeline style invalidation, or a safe performance fix.

// @vitest-environment jsdom

import { act, cleanup, render } from "@testing-library/react";
import { QueryClient, QueryClientProvider } from "@tanstack/react-query";
import { MemoryRouter } from "react-router-dom";
import { afterEach, describe, expect, it } from "vitest";
import type { TimelineRow } from "@bb/server-contract";
import { BottomAnchoredScrollBody } from "@/components/ui/bottom-anchored-scroll-body";
import { installFakeFlowLayout } from "@/test/fake-flow-layout";
import { conversationRow, toolRow } from "@/test/fixtures/thread-timeline-rows";
import { ThreadTimelineRows } from "./ThreadTimelineRows";

const VIEWPORT_HEIGHT = 600;
const ROW_HEIGHT = 140;
const LAYOUT_GETTERS = [
  [Element.prototype, "scrollTop"],
  [Element.prototype, "scrollLeft"],
  [Element.prototype, "scrollHeight"],
  [Element.prototype, "scrollWidth"],
  [Element.prototype, "clientHeight"],
  [Element.prototype, "clientWidth"],
  [Element.prototype, "clientTop"],
  [Element.prototype, "clientLeft"],
  [HTMLElement.prototype, "offsetHeight"],
  [HTMLElement.prototype, "offsetWidth"],
  [HTMLElement.prototype, "offsetTop"],
  [HTMLElement.prototype, "offsetLeft"],
] as const;
const LAYOUT_METHODS = [
  [Element.prototype, "getBoundingClientRect"],
  [Element.prototype, "getClientRects"],
] as const;

const restores: (() => void)[] = [];

afterEach(() => {
  cleanup();
  for (const restore of restores.splice(0).reverse()) restore();
  document.body.replaceChildren();
});

function recordLayoutReads() {
  const reads: string[] = [];
  const stacks = new Set<string>();
  let recording = false;
  const note = (name: string) => {
    if (recording) {
      reads.push(name);
      stacks.add(
        `${name}\n${new Error().stack?.split("\n").slice(3, 7).join("\n")}`,
      );
    }
  };
  for (const [prototype, name] of LAYOUT_GETTERS) {
    const descriptor = Object.getOwnPropertyDescriptor(prototype, name);
    const get = descriptor?.get;
    if (descriptor === undefined || get === undefined) continue;
    Object.defineProperty(prototype, name, {
      ...descriptor,
      get(this: Element) {
        note(name);
        return get.call(this);
      },
    });
    restores.push(() => Object.defineProperty(prototype, name, descriptor));
  }
  for (const [prototype, name] of LAYOUT_METHODS) {
    const original = prototype[name];
    Object.defineProperty(prototype, name, {
      configurable: true,
      writable: true,
      value(this: Element) {
        note(name);
        return Reflect.apply(original, this, []);
      },
    });
    restores.push(() =>
      Object.defineProperty(prototype, name, {
        configurable: true,
        writable: true,
        value: original,
      }),
    );
  }
  const originalGetComputedStyle = window.getComputedStyle;
  window.getComputedStyle = (...args) => {
    note("getComputedStyle");
    return originalGetComputedStyle(...args);
  };
  restores.push(() => {
    window.getComputedStyle = originalGetComputedStyle;
  });
  return {
    stacks: () => [...stacks],
    during(run: () => void): string[] {
      reads.length = 0;
      stacks.clear();
      recording = true;
      try {
        run();
      } finally {
        recording = false;
      }
      return [...reads];
    },
  };
}

function historyRows(count: number, firstSeq = 1): TimelineRow[] {
  return Array.from({ length: count }, (_, offset): TimelineRow => {
    const seq = firstSeq + offset;
    if (seq % 3 === 0) {
      return toolRow({
        id: `row-${seq}`,
        seq,
        toolArgs: { path: `src/file-${seq}.ts`, limit: 200, offset: seq },
        toolName: "Read",
        turnId: `turn-${seq}`,
      });
    }
    return conversationRow({
      id: `row-${seq}`,
      role: seq % 3 === 1 ? "user" : "assistant",
      seq,
      text:
        seq % 3 === 1
          ? `Question ${seq}\n\n${"A line of the question.\n\n".repeat(24)}`
          : `Answer ${seq}\n\n| a | b |\n| - | - |\n| ${seq} | two |\n\n\`\`\`ts\nconst value = ${seq};\n\`\`\``,
      turnId: `turn-${seq}`,
    });
  });
}

function setup(rows: TimelineRow[]) {
  const layout = installFakeFlowLayout({
    heightOf: (element) =>
      element.hasAttribute("data-timeline-row-id") &&
      element.parentElement?.hasAttribute("data-timeline-items") === true
        ? ROW_HEIGHT
        : null,
    scrollElement: () =>
      document.querySelector<HTMLElement>("[data-page-scroll-viewport]"),
    viewportHeight: VIEWPORT_HEIGHT,
  });
  restores.push(layout.restore);
  const recorder = recordLayoutReads();
  const queryClient = new QueryClient();
  const tree = (timelineRows: TimelineRow[], timelineKey = "initial") => (
    <QueryClientProvider client={queryClient}>
      <MemoryRouter>
        <BottomAnchoredScrollBody
          key={timelineKey}
          footer={null}
          maxWidthClassName="max-w-3xl"
        >
          <ThreadTimelineRows
            timelineRows={timelineRows}
            threadRuntimeDisplayStatus="idle"
          />
        </BottomAnchoredScrollBody>
      </MemoryRouter>
    </QueryClientProvider>
  );
  const view = render(tree(rows));
  for (let frame = 0; frame < 3; frame += 1) act(() => layout.frame());
  const mounted = () =>
    [
      ...document.querySelectorAll<HTMLElement>(
        "[data-timeline-items] > [data-timeline-row-id]",
      ),
    ].map((row) => row.dataset.timelineRowId ?? "");
  return {
    layout,
    mounted,
    recorder,
    rerender: (timelineRows: TimelineRow[], timelineKey?: string) =>
      view.rerender(tree(timelineRows, timelineKey)),
  };
}

describe("timeline rows mounting", () => {
  it.each([12, 150, 300])(
    "records warm remount layout requests for %i rows",
    (count) => {
      const rows = historyRows(count);
      const harness = setup(rows);
      for (const timelineKey of ["second", "first", "second", "first"]) {
        let committed: string[] = [];
        const reads = harness.recorder.during(() => {
          act(() => harness.rerender(rows, timelineKey));
          committed = harness.mounted();
          for (let frame = 0; frame < 3; frame += 1)
            act(() => harness.layout.frame());
        });
        const counts = Object.fromEntries(
          [...new Set(reads)].map((name) => [
            name,
            reads.filter((read) => read === name).length,
          ]),
        );
        console.log(
          JSON.stringify({
            count,
            timelineKey,
            committedRows: committed.length,
            settledRows: harness.mounted().length,
            counts,
          }),
        );
        if (count === 300 && timelineKey === "second")
          console.log(JSON.stringify({ stacks: harness.recorder.stacks() }));
        expect(harness.mounted()).toContain(`row-${count}`);
      }
    },
  );

  it("reads no layout while a scroll mounts new rows, and covers the viewport by the next paint", () => {
    const harness = setup(historyRows(150));
    expect(harness.mounted()).not.toContain("row-1");

    act(() => harness.layout.userScrollTo(0));
    expect(harness.mounted()).not.toContain("row-4");
    const reads = harness.recorder.during(() =>
      act(() => harness.layout.runFrameCallbacks()),
    );
    act(() => harness.layout.flushResizeObservers());

    expect(harness.mounted()).toEqual(
      expect.arrayContaining(["row-1", "row-2", "row-3", "row-4"]),
    );
    expect(reads).toEqual([]);
  });

  it("reads no layout while a streamed row mounts at the end", () => {
    const rows = historyRows(150);
    const harness = setup(rows);
    const grown = [...rows, ...historyRows(1, 151)];

    const reads = harness.recorder.during(() =>
      act(() => harness.rerender(grown)),
    );

    expect(harness.mounted()).toContain("row-151");
    expect(reads).toEqual([]);
  });
});

5. Root cause: known lifetime, unresolved latency

Page lifetime restarts when thread identity changes

ThreadScreen keys the page shell by thread identity. React therefore recreates its subtree when switching thread IDs. Reusing timeline query data does not preserve the component instances or their observers across that key change. The probe models this lifetime one layer below PageShell.

Scroll/window geometry reads remain reachable on remount

Recorded stacks point to connect → syncFloor / reconcile, naturalFloorHeight, and syncWindow viewport/container rectangles. Resize-observer settlement also invokes reconciliation. Geometry reads are real core requests, but whether a given read forces expensive browser work depends on preceding invalidations, which this test cannot observe.

Table ancestor discovery accounts for the observed style requests

The getComputedStyle stacks point to readMarkdownTableGeometry and findHorizontalClipAncestor. The fixture includes Markdown tables. JSDOM does not load the application's normal CSS layout, so clipping-ancestor discovery and its counter totals are not a quantitative proxy for the browser. The nullish-assignment cache can also repeat a no-ancestor search in this synthetic environment. No claim is made that these style requests cause 78 separate browser recalculations.

function findHorizontalClipAncestor(element: HTMLElement): HTMLElement | null {
  let current: HTMLElement | null = element;
  while (current && current !== document.body) {
    if (
      HORIZONTAL_CLIP_OVERFLOW_VALUES.has(getComputedStyle(current).overflowX)
    ) {
      return current;
    }
    current = current.parentElement;
  }
  return null;

The existing shared resize dispatcher already batches reads before writes. A blanket claim that these observer callbacks are unbatched is not supported. Both existing tests asserting no synchronous layout reads while mounting additional rows during scroll or streaming passed. The root cause of the reported latency is still unknown.

6. Proposed next experiment and fix decision

Capture a real-browser trace of warm sidebar switches on trusted main, then identify read-after-write invalidation stacks before changing scroll anchoring, observer batching, or view lifetime. Use two synthetic long threads in a fresh isolated app, warm them, alternate actual sidebar clicks, and capture Performance.getMetrics plus a timeline trace. Distinguish total CPU from longest task, and separate initial mount, ResizeObserver delivery, table measurement, composer sizing, and table-of-contents work. Compare with and without third-party frontend assets only after verifying the browser page and cached histories are equivalent.

No production fix or pull request is safe here: the reported browser defect has not been reproduced and there is no focused pre-fix failing regression test for its latency. Removing geometry reads speculatively could break initial positioning, restored anchors, or viewport coverage. The diagnostic passes on unchanged production code and must not be presented as proof of a performance repair. No branch, commit, or PR was created in the target repository.
