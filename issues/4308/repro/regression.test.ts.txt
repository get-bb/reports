  it("rejects unsolicited focus from an uncontrolled hidden browser", async () => {
    vi.useFakeTimers();
    const focusHostWebContents = vi.fn();
    const manager = createDesktopBrowserViewManager({ focusHostWebContents });
    const hostWindow = new FakeHostWindow({
      contentBounds: { width: 700, height: 450 },
      webContentsId: 79,
    });
    try {
      attachBrowserTab({
        manager,
        hostWindow,
        tabId: "browser:a",
        url: "https://example.com",
      });
      await vi.advanceTimersByTimeAsync(0);
      manager.setVisible({
        hostWindow,
        request: { tabId: "browser:a", visible: false },
      });
      const guest = requireFakeView(0).webContents;
      guest.emitDidNavigate("https://example.com/next");
      guest.nativelyFocused = true;
      guest.emitFocus();
      await vi.advanceTimersByTimeAsync(0);
      expect.soft(focusHostWebContents).toHaveBeenCalledExactlyOnceWith(79);
      expect.soft(hostWindow.webContents.sentChannels).not.toContain(
        "bb-desktop:browser:focused",
      );
    } finally {
      manager.destroyAll();
      vi.useRealTimers();
    }
  });
