import { createThreadSection, getThread } from "@bb/db";
import { threadListResponseSchema } from "@bb/server-contract";
import { describe, expect, it } from "vitest";
import { readJson } from "../helpers/json.js";
import {
  seedHostSession,
  seedProjectWithSource,
  seedThread,
} from "../helpers/seed.js";
import { withTestHarness } from "../helpers/test-app.js";

describe("issue 7919 hidden fork placement", () => {
  it.each(["pin", "section"] as const)(
    "makes an accepted %s placement discoverable",
    async (action) => {
      await withTestHarness(async (harness) => {
        const { host } = seedHostSession(harness.deps);
        const { project } = seedProjectWithSource(harness.deps, {
          hostId: host.id,
          path: "/tmp/issue-7919-source",
        });
        const source = seedThread(harness.deps, { projectId: project.id });
        const fork = seedThread(harness.deps, {
          projectId: project.id,
          sourceThreadId: source.id,
          lifecycleOwnerThreadId: source.id,
          originKind: "fork",
          originPluginId: "side-chat",
          visibility: "hidden",
        });
        const section = createThreadSection(harness.db, harness.hub, {
          name: "Reproduction",
        });
        if (section.status !== "created")
          throw new Error("Section creation failed");
        const response = await harness.app.request(
          `/api/v1/threads/${fork.id}${action === "pin" ? "/pin" : ""}`,
          action === "pin"
            ? { method: "POST" }
            : {
                method: "PATCH",
                headers: { "content-type": "application/json" },
                body: JSON.stringify({ sectionId: section.section.id }),
              },
        );
        expect(response.status).toBe(200);
        const saved = getThread(harness.db, fork.id);
        expect(saved?.sourceThreadId).toBe(source.id);
        if (action === "pin")
          expect(saved?.pinnedAt).toEqual(expect.any(Number));
        else expect(saved?.sectionId).toBe(section.section.id);
        const listPath =
          action === "pin"
            ? "/api/v1/threads"
            : `/api/v1/threads?sectionId=${section.section.id}`;
        const list = threadListResponseSchema.parse(
          await readJson(await harness.app.request(listPath)),
        );
        const listedBeforePromotion = list.some(
          (thread) => thread.id === fork.id,
        );
        const promotion = await harness.app.request(
          `/api/v1/threads/${fork.id}`,
          {
            method: "PATCH",
            headers: { "content-type": "application/json" },
            body: JSON.stringify({ visibility: "visible" }),
          },
        );
        expect(promotion.status).toBe(200);
        const promotedList = threadListResponseSchema.parse(
          await readJson(await harness.app.request(listPath)),
        );
        const listedAfterPromotion = promotedList.some(
          (thread) => thread.id === fork.id,
        );
        console.log(
          JSON.stringify({
            action,
            status: response.status,
            visibility: saved?.visibility,
            metadataSaved: true,
            listedBeforePromotion,
            listedAfterPromotion,
            sourcePreserved:
              getThread(harness.db, fork.id)?.sourceThreadId === source.id,
          }),
        );
        expect(listedAfterPromotion).toBe(true);
        expect(listedBeforePromotion).toBe(true);
      });
    },
  );
});

5. Root cause

The hidden state is intentional: side-chat creation requests a hidden fork with a lifecycle owner and source. The mismatch occurs when this fork is presented to ordinary thread-list actions.

  1. Core Pin binds without a hidden-state guard and calls the standard SDK pin operation.
  2. Move to section guards child/archive state, not hidden state. Its successful path updates sectionId and may unpin; it does not change visibility.
  3. The pin route accepts the hidden public thread. pinThread writes only timestamp, sort key, and update time, not visibility.
  4. The section mutation treats sectionId and visibility as separate optional patches.
  5. Pinned roots require visible; normal list queries require visible unless includeHidden is explicit. Persisted placement cannot overcome those filters.
Pin mutation writes:
  pinnedAt: args.pinnedAt ?? now
  pinSortKey
  updatedAt: now

Pinned root filter requires:
  eq(threads.visibility, "visible")

Default thread/section filter requires:
  options.includeHidden ? undefined : eq(threads.visibility, "visible")

The deeper action contract also loses the necessary state: toThreadActionTarget does not copy visibility, and the public PluginThreadActionTarget contract does not declare it. The action implementations cannot simply inspect an existing visibility field.

6. Proposed fix and simple-fix gate

Choose the product policy first: either explicitly promote an eligible side chat when a user chooses a placement, or suppress/disable these actions for hidden threads and supply a separate promotion action. Preserve the source reference and decide lifecycle ownership deliberately. Do not globally promote every hidden thread during ordinary metadata updates: other plugins also use hidden threads, and existing public API coverage intentionally allows hidden section assignments.

No production fix or PR: the minimal alternatives require a product decision; the hide/disable approach also needs a change to the public shared action-target contract, while promotion affects core, thread-list, and side-chat semantics. Those fail the authorized simple-fix constraints. No open PR was connected to this issue in timeline metadata or the open-PR search at investigation time. No branch was pushed.
