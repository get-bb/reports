Full reproduction test
import { publicApiRoutes } from "@bb/server-contract";
import { expect, it } from "vitest";
import { registerHostRpcResponder } from "../helpers/host-rpc.js";
import { seedThreadFixture } from "../helpers/seed.js";
import { withTestHarness } from "../helpers/test-app.js";

it("records raw HTML and the removed legacy content endpoint on main", async () => {
  await withTestHarness(async (harness) => {
    const { host, session, thread } = seedThreadFixture(harness);
    const storageRoot = `/tmp/bb-host-data/${host.id}/thread-storage/${thread.id}`;
    const source = "<!doctype html><html><body><h1>Route probe</h1></body></html>";
    const bytes = Buffer.from(source);
    registerHostRpcResponder(harness, {
      hostId: host.id,
      sessionId: session.id,
      handle: ({ command }) => {
        if (command.type !== "host.read_file_chunk") {
          throw new Error(`Unexpected command: ${command.type}`);
        }
        expect(command.rootPath).toBe(storageRoot);
        expect(command.path).toBe(`${storageRoot}/preview.html`);
        return {
          ok: true,
          result: {
            path: command.path,
            content: bytes
              .subarray(command.offset, command.offset + command.length)
              .toString("base64"),
            offset: command.offset,
            mimeType: "text/html",
            modifiedAtMs: 1234,
            sizeBytes: bytes.length,
            revision: "0".repeat(64),
          },
        };
      },
    });
    const rawPath = publicApiRoutes.threads.storageFile.path
      .replace(":id", thread.id)
      .replace(":filePath{.+}", "preview.html");
    const raw = await harness.app.request(`/api/v1${rawPath}`);
    const body = await raw.text();
    expect(raw.status).toBe(200);
    expect(body).toBe(source);
    expect(raw.headers.get("content-security-policy")).toBe("sandbox allow-scripts");
    expect(raw.headers.get("cache-control")).toBe("no-store");
    const legacy = await harness.app.request(
      `/api/v1/threads/${thread.id}/thread-storage/content?path=preview.html`,
    );
    expect(legacy.status).toBe(404);
    await legacy.text();
    console.log(JSON.stringify({
      rawStatus: raw.status,
      sourceUnchanged: body === source,
      sandbox: raw.headers.get("content-security-policy"),
      cacheControl: raw.headers.get("cache-control"),
      legacyContentStatus: legacy.status,
    }));
  });
});

5. Root cause and scope

The report's camera-aware integration is outside the tested upstream implementation. Main has one raw storage-file owner, and it streams host bytes through a sandboxed response. There is no camera-aware handler in the thread module for the first registration to shadow.

  • Server registration order: general file routes do precede thread routes; order alone does not establish a duplicate handler.
  • Only raw storage-file registration: resolves the thread storage root and calls the common raw file server.
  • Common serving implementation: streams the document rather than constructing camera markup.
  • Public route contract: raw storage files declare no query request; no camera contract is shipped here.
  • Current HTML renderer: a sandboxed iframe with loading callbacks, not the reported local camera implementation.
  • Inline visualization renderer: likewise loads a sandboxed document directly.
