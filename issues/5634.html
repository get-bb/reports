#5634 · Terminal relay shares the server event loop

Bug Priority: Medium Effort: High host ui perf · GitHub issue · 2026-10-16 · base 399d63d27702638622ed6522bf874808a364fb40

Verdict: REPRODUCED · Root-cause confidence: high · Label: confirmed-repro

1. TL;DR

A terminal key can wait behind unrelated synchronous work in the server before its echo returns. A fresh probe using the real server WebSocket routes, real migrated SQLite, a separate client process, and an actual PTY reproduced approximately 903 ms of echo latency when input arrived 100 ms into a one-second server-thread stall. A separate delayed TCP connection added approximately 80 ms to echo latency, as expected for a round trip. The server data relay runs on the same event loop as product work; removing individual slow operations does not make that relay independent. Immediate predictive drawing would be new client behavior and is not approved by this report.

Complete separate-process PTY and WebSocket fixture: slopcop-5634-client.mjs
import { createRequire } from "node:module";
import { createServer, connect } from "node:net";
import { once } from "node:events";
import { resolve } from "node:path";
import { setTimeout as sleep } from "node:timers/promises";

const serverRequire = createRequire(resolve("apps/server/package.json"));
const hostRequire = createRequire(resolve("apps/host-daemon/package.json"));
const WebSocket = serverRequire("ws");
const { spawn } = hostRequire("node-pty");
let daemon;
let browser;
let terminal;
let terminalId;
let serverUrl;
let nextSeq = 0;
let pending;
let proxy;
const sockets = new Set();

async function openBrowser(oneWayMs) {
  if (browser) {
    browser.close();
    await once(browser, "close");
  }
  if (proxy) {
    for (const socket of sockets) socket.destroy();
    proxy.close();
    proxy = null;
  }
  const address = new URL(serverUrl);
  let port = address.port;
  if (oneWayMs > 0) {
    proxy = createServer((downstream) => {
      const upstream = connect(Number(address.port), "127.0.0.1");
      for (const socket of [upstream, downstream]) {
        sockets.add(socket);
        socket.on("close", () => sockets.delete(socket));
        socket.on("error", () => {});
      }
      for (const [source, destination] of [[upstream, downstream], [downstream, upstream]]) {
        source.on("data", (bytes) => setTimeout(() => {
          if (!destination.destroyed) destination.write(bytes);
        }, oneWayMs));
        source.on("end", () => setTimeout(() => destination.end(), oneWayMs));
      }
    });
    proxy.listen(0, "127.0.0.1");
    await once(proxy, "listening");
    port = proxy.address().port;
  }
  browser = new WebSocket(`ws://127.0.0.1:${port}/ws/terminals/${terminalId}`);
  await new Promise((resolveAttached, reject) => {
    browser.on("error", reject);
    browser.on("message", (raw) => {
      const message = JSON.parse(raw.toString());
      if (message.type === "attached") resolveAttached();
      if (message.type === "error") reject(new Error(message.message));
      if (message.type === "output" && pending) {
        pending.text += Buffer.from(message.chunk.dataBase64, "base64").toString();
        if (pending.text.includes(pending.key)) {
          const current = pending;
          pending = null;
          clearTimeout(current.timeout);
          current.resolve({
            echo: current.key,
            latencyMs: Math.round((performance.now() - current.started) * 10) / 10,
          });
        }
      }
    });
  });
}

async function initialize(message) {
  serverUrl = message.url;
  terminalId = message.terminalId;
  terminal = spawn("/bin/cat", [], {
    name: "xterm-256color",
    cols: 80,
    rows: 24,
    cwd: message.cwd,
    env: { PATH: "/usr/bin:/bin", TERM: "xterm-256color", LANG: "C.UTF-8" },
  });
  daemon = new WebSocket(`${serverUrl.replace("http:", "ws:")}/internal/ws?sessionId=${message.sessionId}`, "bb-host-daemon.v1", {
    headers: { Authorization: `Bearer ${message.hostKey}` },
  });
  daemon.on("message", (raw) => {
    const command = JSON.parse(raw.toString());
    if (command.type === "terminal.attach") {
      daemon.send(JSON.stringify({
        type: "terminal.replay",
        requestId: command.requestId,
        terminalId,
        chunks: [],
        replayStartSeq: nextSeq,
        nextSeq,
      }));
    }
    if (command.type === "terminal.input") {
      terminal.write(Buffer.from(command.dataBase64, "base64").toString());
    }
  });
  terminal.onData((text) => {
    const bytes = Buffer.from(text);
    daemon.send(JSON.stringify({
      type: "terminal.output",
      terminalId,
      chunk: { seq: nextSeq, dataBase64: bytes.toString("base64") },
    }));
    nextSeq += bytes.length;
  });
  await once(daemon, "open");
  await openBrowser(0);
  return { ready: true };
}

async function measure(message) {
  await sleep(message.delayMs);
  return new Promise((resolveEcho, reject) => {
    pending = {
      key: message.key,
      text: "",
      started: performance.now(),
      resolve: resolveEcho,
      timeout: setTimeout(() => reject(new Error("No terminal echo within 5 seconds")), 5000),
    };
    browser.send(JSON.stringify({ type: "input", dataBase64: Buffer.from(message.key).toString("base64") }));
  });
}

process.on("message", async (message) => {
  try {
    if (message.action === "init") process.send(await initialize(message));
    if (message.action === "prepare") {
      await openBrowser(message.oneWayMs);
      process.send({ prepared: true });
    }
    if (message.action === "measure") process.send(await measure(message));
    if (message.action === "stop") {
      browser?.terminate();
      daemon?.terminate();
      terminal?.kill();
      for (const socket of sockets) socket.destroy();
      proxy?.close();
      process.send({ stopped: true });
    }
  } catch (error) {
    console.error(error);
    process.send({ error: String(error) });
  }
});

5. Root cause

The two WebSocket routes are installed into the same server instance. Their message callbacks directly invoke the terminal lifecycle and daemon protocol handlers; there is no independently scheduled data plane. A synchronous loop on the server worker prevents those callbacks from handling newly arriving bytes even while the external client and PTY remain schedulable.

  • Browser terminal route and direct input callbacks.
  • Shared daemon route and direct message callback.
  • Input forwarding reads and updates terminal state, emits session updates, sends the daemon input message, and emits a plugin event on this path.
  • Output forwarding reads the terminal row and forwards its output chunk in the daemon message handler.
  • Input conversion only calls the supplied input callback; the xterm integration separately receives output and forwards keystrokes.

The database work is part of the serial path, but this experiment does not attribute the injected stall to a slow database query. The stall is deliberately unrelated work in the same event loop. Caching terminal rows or shrinking one slow task cannot guarantee continuous I/O while that event loop is blocked. Separately, without local prediction the client cannot show confirmed PTY echo before the remote trip completes; a renderer replacement alone does not remove this network dependency.

6. Proposed fix and safe-fix gate

Establish an independently serviced terminal-byte path, keeping authorization, terminal lifecycle, database policy, and plugin event policy on the main server. Any design must preserve ordering, replay offsets, pacing/acks, disconnect revocation, session ownership, and daemon compatibility. In particular, moving only browser sockets while keeping incoming daemon bytes on the blocked server loop would not isolate output.

Predictive local echo should be treated as a separate product feature, with explicit rules for passwords, alternate-screen programs, mismatches, settings, SDK, and CLI. This report does not authorize that feature.

No automatic PR: the verified failure is an execution/transport ownership problem. A safe repair requires architecture and potentially public daemon-protocol decisions, so it fails the simple-fix gate even if a small superficial optimization could fit the line limit.
