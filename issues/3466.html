#3466 · SDK response method dispatch

Bug · High priority · Low effort · plugins · 2026-09-11

Base: b04091eca67bb463cce16b9bfa318660bd20ab69 · Issue

REPRODUCED · root-cause confidence: high

TL;DR

The SDK rejects while reading installer events when the server adapter replaces the global Response constructor. The response returned by the transport is native, but the SDK invokes the adapter prototype method on it. That method needs adapter-private state and throws. The same agent reproduced the exact exception in two trusted checkouts; no provider installation or live BB instance was used.

Root cause

SDK response consumption calls the global Response prototype with a transport-owned receiver. Server startup uses Hono serve with default global overrides. The installed adapter text method calls its private getResponseCache symbol, absent on native Fetch responses. Installer route returns newline-delimited data. The parsing schema is unchanged.

Proposed fix and validation

Use the returned response's own text method. Annotate the local response as Response to widen Hono's inferred text return type without a cast or contract change. The focused regression test substitutes a global Response class with private state, supplies a native transport response, and checks multiple events, CRLF and blank lines. Before: 1 failed, 104 passed. After: all 105 SDK tests pass, SDK typecheck and build pass. The real adapter reproduction also passes after rebuilding.

import { createRequire } from "node:module";
import { expect, it, vi } from "vitest";
import { createBbSdk } from "../src/core.js";
import { createHttpTransport } from "../src/transport-http.js";

it("reads native installer response bodies under the installed Hono global adapter", async () => {
  const NativeResponse = globalThis.Response;
  const NativeRequest = globalThis.Request;
  const requireServer = createRequire(new URL("../../../apps/server/package.json", import.meta.url));
  const { getRequestListener } = requireServer("@hono/node-server");
  const events = [
    { type: "started", provider: "synthetic", command: "synthetic-installer" },
    { type: "output", provider: "synthetic", stream: "stdout", text: "synthetic-output" },
    { type: "completed", provider: "synthetic", exitCode: 0, signal: null, success: true },
  ];
  const run = (response: Response) => createBbSdk({ transport: createHttpTransport({ baseUrl: "http://fixture.invalid", runtime: "node", fetch: async () => response }) }).hosts.installProviderCli({ hostId: "synthetic", provider: "synthetic", actionKind: "update" });
  const outcomes: Record<string, unknown> = {};
  try {
    getRequestListener(() => new NativeResponse("unused"));
    expect(globalThis.Response).not.toBe(NativeResponse);
    vi.stubGlobal("Response", globalThis.Response);
    let legacyError = "";
    try { await globalThis.Response.prototype.text.call(new NativeResponse("control")); }
    catch (error) { if (!(error instanceof Error)) throw error; legacyError = error.message; }
    expect(legacyError).toContain("getResponseCache");
    outcomes.legacyDispatchControl = legacyError;
    const response = new NativeResponse("\r\n" + events.map(e => JSON.stringify(e)).join("\r\n\r\n") + "\r\n");
    expect(await run(response)).toEqual(events);
    expect(response.bodyUsed).toBe(true);
    outcomes.success = { events: events.length, bodyUsed: response.bodyUsed };
    expect(await run(new NativeResponse(" \r\n\n"))).toEqual([]);
    outcomes.blankBody = [];
    const event = { type: "error", provider: "synthetic", message: "synthetic installer error" };
    expect(await run(new NativeResponse(JSON.stringify(event)))).toEqual([event]);
    outcomes.installerErrorEvent = "returned as event";
    await expect(run(new NativeResponse("{"))).rejects.toBeInstanceOf(SyntaxError);
    outcomes.invalidJson = "SyntaxError";
    await expect(run(new NativeResponse('{"type":"unrecognized"}'))).rejects.toMatchObject({ name: "ZodError" });
    outcomes.invalidEvent = "ZodError";
    const broken = new NativeResponse(new ReadableStream({ start(c) { c.error(new Error("synthetic body failure")); } }));
    await expect(run(broken)).rejects.toThrow("synthetic body failure");
    outcomes.bodyFailure = "synthetic body failure";
    await expect(run(new NativeResponse(JSON.stringify({ error: "synthetic unavailable" }), { status: 503, headers: { "content-type": "application/json" } }))).rejects.toMatchObject({ status: 503 });
    outcomes.httpFailure = 503;
    console.log("ISSUE3466_EVIDENCE=" + JSON.stringify(outcomes));
  } finally {
    vi.unstubAllGlobals();
    globalThis.Response = NativeResponse;
    globalThis.Request = NativeRequest;
  }
});
