import { fork } from 'node:child_process';
import { once } from 'node:events';
import { pathToFileURL } from 'node:url';
import { resolve } from 'node:path';
import assert from 'node:assert/strict';
const { createChildChannel } = await import(pathToFileURL(resolve(process.argv[2], 'packages/host-watcher/src/parcel-subprocess/fork-channel.ts')));
for (let i = 0; i < 20; i++) {
  const child = fork(new URL('./idle.mjs', import.meta.url), [], { stdio: ['ignore', 'ignore', 'ignore', 'ipc'], execArgv: [] });
  const channel = createChildChannel(child);
  let exits = 0;
  channel.onExit(() => exits++);
  await once(child, 'spawn');
  const closed = once(child, 'close');
  channel.kill();
  await closed;
  assert.equal(exits, 1);
  assert.throws(() => process.kill(child.pid, 0), { code: 'ESRCH' });
}
console.log('PASS: 20 watcher-channel children closed; exit notified once; all PIDs absent.');

Root cause

Not established. createForkChannel uses Node fork directly with IPC; it does not specify a shell. createChildChannel registers exit/error listeners and sends SIGKILL when killed. The test exercises this real adapter with a minimal Node child. It cannot exclude shell grandchildren in ACP providers or a packaged Linux-specific issue.
