import { describe, expect, it } from "vitest";
import { createClaudeDeltaHarness } from "./delta-test-harness.js";

function startTurn(
  harness: ReturnType<typeof createClaudeDeltaHarness>,
  threadId: string,
) {
  harness.translate(
    {
      jsonrpc: "2.0",
      method: "sdk/message",
      params: {
        threadId,
        message: {
          type: "assistant",
          message: { id: threadId, content: [] },
        },
      },
    },
    { threadId },
  );
}

function resultEvent(threadId: string, failed: boolean) {
  return {
    jsonrpc: "2.0",
    method: "sdk/message",
    params: {
      threadId,
      message: {
        type: "result",
        subtype: failed ? "error_during_execution" : "success",
        is_error: failed,
        ...(failed ? { api_error_status: 402 } : {}),
        result: failed ? "Synthetic Claude billing rejection" : "ok",
        usage: {},
        modelUsage: {},
      },
    },
  };
}

describe("issue 7453 provider-boundary investigation", () => {
  it("accepts a successful Claude result without a billing error", () => {
    const harness = createClaudeDeltaHarness();
    startTurn(harness, "claude-success");
    const events = harness.translate(resultEvent("claude-success", false), {
      threadId: "claude-success",
    });
    expect(events.filter((event) => event.type === "provider/error")).toEqual([]);
    expect(events).toContainEqual(
      expect.objectContaining({ type: "turn/completed", status: "completed" }),
    );
  });

  it("surfaces the Claude upstream billing rejection verbatim", () => {
    const harness = createClaudeDeltaHarness();
    startTurn(harness, "claude-billing");
    const events = harness.translate(resultEvent("claude-billing", true), {
      threadId: "claude-billing",
    });
    expect(events.filter((event) => event.type === "provider/error")).toEqual([
      expect.objectContaining({
        detail: "Synthetic Claude billing rejection",
        errorInfo: {
          category: "billing",
          providerCode: "error_during_execution",
          httpStatusCode: 402,
        },
      }),
    ]);
    expect(events).toContainEqual(
      expect.objectContaining({ type: "turn/completed", status: "failed" }),
    );
  });

  it("does not carry a billing failure into another Claude thread", () => {
    const harness = createClaudeDeltaHarness();
    startTurn(harness, "claude-billing");
    harness.translate(resultEvent("claude-billing", true), {
      threadId: "claude-billing",
    });
    startTurn(harness, "claude-other");
    const events = harness.translate(resultEvent("claude-other", false), {
      threadId: "claude-other",
    });
    expect(events.filter((event) => event.type === "provider/error")).toEqual([]);
    expect(events).toContainEqual(
      expect.objectContaining({ type: "turn/completed", status: "completed" }),
    );
  });
});

5. Root cause and limits

Root cause of the reported failure: unknown. The evidence does not show that bb incorrectly uses Codex balance for Claude. It is also insufficient to conclude that the user's authentication is incorrect or that a subscription or API balance is exhausted. Do not mistake an upstream error propagation test for proof that the upstream service caused this specific incident.

  • Result failure and detail extraction: failure is identified from the Claude result; its result string is returned verbatim when present.
  • Provider-error emission: the Claude result supplies the HTTP status, subtype, and displayed error detail.
  • HTTP billing classification and Claude billing classification: HTTP 402 and Claude billing errors classify as billing. These mappings are not shared-credit checks.
  • Claude launch environment and SDK setting sources: Claude receives the supplied environment and reads user/project/local settings. Which authentication and endpoint the reporter actually uses is unobserved.
  • Claude SDK invocation: the bridge calls the Claude SDK with its session options. This is static evidence about this path, not an end-to-end proof about every server/plugin hook.
