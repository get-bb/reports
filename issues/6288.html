#6288 · Context-aware lightbox portal ownership

Bug Priority: Low Effort: Low ui plugins · GitHub issue

2026-10-26 · trusted origin/main: 3a2c49a267cd2a62a96ebff880068a7c37074251

Verdict: REPRODUCED · Root-cause confidence: high

1. TL;DR

The image preview is rendered in a portal, outside its original DOM parent. Its portal incorrectly identifies itself as an anonymous plugin root, both for ordinary host content and for content owned by a named plugin. A component-level reproduction confirms both attribute failures on trusted main, twice in separate clean checkouts. The underlying import selects the standalone shared-UI hook instead of the host's context-aware hook. This permits matching of the anonymous-root CSS selector; a visible styling collision, desktop behavior, mobile behavior, and release-package behavior were not tested.

// @vitest-environment jsdom

import type { ReactNode } from "react";
import { cleanup, fireEvent, render, screen } from "@testing-library/react";
import { afterEach, describe, expect, it } from "vitest";
import { PluginContext } from "@/components/plugin/plugin-context";
import { ImageLightbox } from "@/components/ui/image-lightbox";
import { Dialog, DialogContent, DialogTitle } from "@bb/shared-ui/dialog";
import {
  Tooltip,
  TooltipContent,
  TooltipProvider,
  TooltipTrigger,
} from "@bb/shared-ui/tooltip";
import {
  Select,
  SelectContent,
  SelectItem,
  SelectTrigger,
  SelectValue,
} from "@bb/shared-ui/select";
import {
  ContextMenu,
  ContextMenuContent,
  ContextMenuItem,
  ContextMenuTrigger,
} from "@bb/shared-ui/context-menu";

function inPluginScope(children: ReactNode) {
  return (
    <PluginContext.Provider value="test-plugin">
      {children}
    </PluginContext.Provider>
  );
}

function openDialog() {
  return (
    <Dialog open>
      <DialogContent>
        <DialogTitle>hi</DialogTitle>
      </DialogContent>
    </Dialog>
  );
}

afterEach(cleanup);

describe("usePortalScopeProps", () => {
  it.each([null, "test-plugin"])(
    "preserves lightbox portal ownership for plugin %s",
    (pluginId) => {
      const { container } = render(
        <PluginContext.Provider value={pluginId}>
          <ImageLightbox
            imageSrc="/scope-check.png"
            imageAlt="Scope check"
            title="Scope check"
            onClose={() => {}}
          />
        </PluginContext.Provider>,
      );

      const dialog = screen.getByRole("dialog", { name: "Scope check" });
      expect(container.contains(dialog)).toBe(false);
      expect(dialog.getAttribute("data-bb-portaled-overlay")).toBe("");
      expect({
        root: dialog.getAttribute("data-bb-plugin-root"),
        plugin: dialog.getAttribute("data-bb-plugin"),
      }).toEqual({ root: pluginId === null ? null : "", plugin: pluginId });
    },
  );

  it("stamps portaled dialog content + overlay inside a plugin slot", () => {
    const { baseElement } = render(inPluginScope(openDialog()));

    const content = baseElement.querySelector('[role="dialog"]');
    expect(content).not.toBeNull();
    expect(content!.getAttribute("data-bb-portaled-overlay")).toBe("");
    expect(content!.getAttribute("data-bb-plugin-root")).toBe("");

    const scoped = baseElement.querySelectorAll("[data-bb-plugin-root]");
    expect(scoped.length).toBe(2);
    for (const root of scoped) {
      expect(root.getAttribute("data-bb-plugin")).toBe("test-plugin");
    }
    expect(
      baseElement.querySelectorAll("[data-bb-portaled-overlay]").length,
    ).toBe(2);
  });

  it("leaves host-tree dialogs unscoped so plugin CSS cannot match them", () => {
    const { baseElement } = render(openDialog());

    const content = baseElement.querySelector('[role="dialog"]');
    expect(content).not.toBeNull();
    expect(content!.hasAttribute("data-bb-plugin-root")).toBe(false);
    expect(content!.getAttribute("data-bb-portaled-overlay")).toBe("");
    expect(baseElement.querySelectorAll("[data-bb-plugin-root]").length).toBe(
      0,
    );
  });

  it("stamps tooltip content (inline-hook variant) inside a plugin slot", () => {
    const { baseElement } = render(
      inPluginScope(
        <TooltipProvider>
          <Tooltip open>
            <TooltipTrigger>trigger</TooltipTrigger>
            <TooltipContent>tip</TooltipContent>
          </Tooltip>
        </TooltipProvider>,
      ),
    );

    const tip = baseElement.querySelector(
      '[data-bb-plugin-root][role="tooltip"], [role="tooltip"]',
    );
    expect(tip).not.toBeNull();
    const scopedTip = baseElement.querySelector("[data-bb-plugin-root]");
    expect(scopedTip).not.toBeNull();
    expect(scopedTip!.getAttribute("data-bb-plugin")).toBe("test-plugin");
    expect(scopedTip!.getAttribute("data-bb-portaled-overlay")).toBe("");
  });

  it("stamps select content so native drag regions cannot swallow options", () => {
    const { baseElement } = render(
      <Select open value="one">
        <SelectTrigger aria-label="Example select">
          <SelectValue />
        </SelectTrigger>
        <SelectContent>
          <SelectItem value="one">One</SelectItem>
        </SelectContent>
      </Select>,
    );

    expect(
      baseElement
        .querySelector('[role="listbox"]')
        ?.getAttribute("data-bb-portaled-overlay"),
    ).toBe("");
  });

  it("stamps context menu content after a real context-menu gesture", () => {
    const { baseElement } = render(
      <ContextMenu>
        <ContextMenuTrigger>Context target</ContextMenuTrigger>
        <ContextMenuContent>
          <ContextMenuItem>Action</ContextMenuItem>
        </ContextMenuContent>
      </ContextMenu>,
    );

    fireEvent.contextMenu(screen.getByText("Context target"));

    const menu = baseElement.querySelector('[role="menu"]');
    expect(menu?.getAttribute("data-bb-portaled-overlay")).toBe("");
    expect(menu?.classList.contains("z-[70]")).toBe(true);
  });
});

5. Root cause

The app lightbox imports the standalone hook directly:

import { usePortalScopeProps } from "@bb/shared-ui/lib/portal-scope";

That hook uses a compile-time ID and always adds a plugin root:

const pluginId =
  typeof __BB_PLUGIN_ID__ === "string" ? __BB_PLUGIN_ID__ : undefined;
return {
  "data-bb-portaled-overlay": "",
  "data-bb-plugin-root": "",
  ...(pluginId !== undefined ? { "data-bb-plugin": pluginId } : {}),
};

In this host test environment the compile-time plugin ID is absent, so the result is an anonymous root. React's PluginContext cannot affect this implementation. The app's environment seam substitutes the host implementation only for importers inside packages/shared-ui. The lightbox lives inside apps/app and is excluded.

The existing host implementation reads PluginContext and returns only the overlay marker when that context is null, or the overlay marker, plugin root and owning ID otherwise. That is the existing behavior already exercised by the dialog cases.

The utility scope root selector contains an anonymous-root alternative:

[data-bb-plugin="PLUGIN_ID"], [data-bb-plugin-root]:not([data-bb-plugin])

Consequently, the incorrect host root is eligible for utility rules from plugins other than an intended owner. This is a CSS ownership defect; no authentication, authorization, data exposure, or exploit was demonstrated.

6. Proposed fix and verification

Use @/lib/portal-scope in the app's lightbox. This reuses existing policy rather than broadening the resolver or changing plugin packaging. Preserve the two ownership cases in the canonical portal-scope suite.

The investigator applied that one-import repair locally. The focused suite passed: 3 files, 12 tests, including lightbox keyboard/dismissal interactions and the existing portal cases. Turbo lint/typecheck passed (246 lint warnings, zero errors). The final change is two files, 26 additions and one deletion: 27 changed text lines. No dependency, lockfile, generated output, public API, protocol, stored data, or permission behavior changes.
